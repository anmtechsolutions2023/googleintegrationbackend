
 SELECT 
    ut.tenant_id,
    ut.user_email,
    ut.is_admin,
    f.feature_name,
    f.scope,
    f.feature_short_name
FROM user_tenants ut
LEFT JOIN tenant_features tf ON ut.id = tf.user_tenants_id
LEFT JOIN features f ON tf.feature_id = f.feature_id
where f.is_active  = TRUE AND tf.is_active = TRUE AND ut.is_active =TRUE
AND ut.tenant_id = 'e3845e08-dcc2-11f0-8e78-0242ac110002' AND ut.user_email = 'anmtechsolutions2023@gmail.com'


 SELECT 
    ut.tenant_id AS "Tenant ID",
    ut.user_email AS "User Email",
    ut.is_admin AS "Is Admin",
    f.feature_name AS "Feature Name",
    f.scope AS "Access Level",
    CASE 
        WHEN ut.is_active = 1 AND tf.is_active = 1 THEN 'Active'
        ELSE 'Disabled'
    END AS "Current Status"
FROM user_tenants ut
LEFT JOIN tenant_features tf ON ut.id = tf.user_tenants_id
LEFT JOIN features f ON tf.feature_id = f.feature_id
